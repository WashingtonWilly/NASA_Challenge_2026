2. Formulación del motor agronómico e índice IRR

ET0 = 0.0023 * (T_med + 17.8) * sqrt(T_max - T_min) * (0.408 * R_a) * dias_mes
P_ef(m) = f_pend(SRTM) * f_FAO(P_m) + Reserva_Andisol(SOC)
ET_c(m) = K_c(etapa) * ET0(m) | Deficit D(m) = max(0, ET_c(m) - P_ef(m))

IRR = w_H2O * I_H2O + w_N * I_N + w_MOS * I_MOS + w_rent * I_rent
I_H2O = 1 - [ sum(D(m)) / sum(ET_c(m)) ]
I_N = clip((sum(Creditos_N) - sum(Extracciones_N) + 300) / 600, 0, 1) + Bono_CN - Penal_Inmov
I_MOS = clip(sum(Residuo_i * h_i * f_forraje) / 4.5, 0, 1) - Penal_Raiz + Bono_Ladera
I_rent = clip(sum(Margen_i) / Max_Margen_Zona, 0, 1) + Bono_Diversificacion
